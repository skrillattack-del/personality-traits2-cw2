namespace = court_maintenance

# Children should acquire personality trait events (supporting 3-9 personality traits with a modal distribution)
court_maintenance.0003 = {
	hidden = yes

	trigger = {
		OR = {
			is_adult = no
			any_courtier_or_guest = {
				is_adult = no
			}
			any_prisoner = {
				is_adult = no
			}
		}
	}

	immediate = {
		if = {
			limit = {
				is_adult = no
				age >= 7
				number_of_personality_traits < personality_trait_limit
			}
			if = {
				limit = {
					age = 9
					has_variable = reincarnation_of
					can_become_reincarnation_trigger = yes
				}
				trigger_event = child_personality.9900
			}
			else = {
				if = {
					limit = {
						number_of_personality_traits < personality_trait_minimum
					}
					trigger_event = {
						on_action = child_personality_gain
						days = { 1 360 }
					}
				}
				else_if = {
					limit = {
						number_of_personality_traits < personality_trait_limit
					}
					random = {
						chance = 50
						trigger_event = {
							on_action = child_personality_gain
							days = { 1 360 }
						}
					}
				}
			}
		}
		every_courtier_or_guest = {
			limit = {
				is_adult = no
				age >= 7
				number_of_personality_traits < personality_trait_limit
			}
			if = {
				limit = {
					number_of_personality_traits < personality_trait_minimum
				}
				trigger_event = {
					on_action = child_personality_gain
					days = { 1 360 }
				}
			}
			else_if = {
				limit = {
					number_of_personality_traits < personality_trait_limit
				}
				random = {
					chance = 50
					trigger_event = {
						on_action = child_personality_gain
						days = { 1 360 }
					}
				}
			}
		}
		every_prisoner = {
			limit = {
				is_adult = no
				age >= 7
				number_of_personality_traits < personality_trait_limit
			}
			if = {
				limit = {
					number_of_personality_traits < personality_trait_minimum
				}
				trigger_event = {
					on_action = child_personality_gain
					days = { 1 360 }
				}
			}
			else_if = {
				limit = {
					number_of_personality_traits < personality_trait_limit
				}
				random = {
					chance = 50
					trigger_event = {
						on_action = child_personality_gain
						days = { 1 360 }
					}
				}
			}
		}
	}
}
